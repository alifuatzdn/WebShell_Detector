<?php
/**
 * Webshell Detector Test - BENIGN
 * * Bu dosya tamamen zararsızdır. Detektörün False Positive (yanlış alarm)
 * oranını ölçmek için benzer kelime kalıpları kullanılarak yazılmıştır.
 */

class SystemUpdateManager {
    private $commandStatus = 'idle';

    // İçinde 'system' kelimesi geçiyor ama bir sınıf metodu, zararlı değil
    public function checkSystemHealth() {
        $this->commandStatus = 'running_diagnostics';
        return ["status" => "OK", "memory_leaks" => false];
    }

    // İçinde 'cmd' kelimesi geçiyor ama sadece statik bir string dizisi
    public function getAllowedCommandsList() {
        $cmd_whitelist = ['backup_db', 'clear_cache', 'generate_report'];
        return $cmd_whitelist;
    }
}

// Tamamen legal bir POST/GET işleme mantığı (Zararlı fonksiyon çağrısı içermez)
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    $action = filter_input(INPUT_POST, 'action', FILTER_SANITIZE_SPECIAL_CHARS);
    
    $manager = new SystemUpdateManager();
    
    if ($action === 'check') {
        $result = $manager->checkSystemHealth();
        echo json_encode($result);
    }
}
?>
